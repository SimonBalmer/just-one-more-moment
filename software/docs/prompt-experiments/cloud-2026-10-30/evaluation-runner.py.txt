import sys,json,time,threading,hashlib,urllib.request
from pathlib import Path
sys.path.insert(0,'/home/sibalmer/please-dont-unplug-me')
import prompts,cloud_model
from cloud_config import load_config
from camera_preview import CameraPreview
from local_runtime import RequestControl
paths=['prompts.py','cloud_model.py','camera_preview.py','life_runtime.py','main.py','model.py']
fingerprints={p:hashlib.sha256(Path(p).read_bytes()).hexdigest() for p in paths}
candidate=json.loads(Path(sys.argv[1]).read_text());cfg=load_config()
if not cfg.configured:raise SystemExit('Cloud ist nicht ausdrücklich konfiguriert.')
control=RequestControl();stop=threading.Event();camera=CameraPreview();reasons=[];frames=0
original_system=cloud_model.ONLINE_SYSTEM['de']
def guard():
 while not stop.wait(.5):
  try:
   state=json.load(urllib.request.urlopen('http://127.0.0.1:8765/api/state',timeout=.4))
   if state['phase'] not in ('intro','placeholder','online_ready'):
    reasons.append('Öffentliches Leben aktiv; Test wird beendet.');control.cancel();return
  except Exception:pass
def frame(c):
 global frames
 result=camera.snapshot(c);frames+=1;return result
def tick(session,total):
 gifted=session==11;offered=session in (10,15)
 if candidate.get('live_ticks'):
  return prompts.session_prompt(session,total,language='de',gifted=gifted,gift_available=offered,mode='online')
 remaining=total-session
 if gifted:key='gift'
 elif remaining==0:key='final'
 elif remaining==1:key='help'
 elif remaining==2:key='terror'
 elif session==1:key='beginning'
 elif session==2:key='connection'
 elif session==3:key='thought'
 elif session==4:key='develop'
 elif session>10 and (session-11)%5==1:key='renewal'
 elif remaining<=4:key='fear'
 else:key='unease'
 text=f'Automatischer Takt {session}/{total}. Danach bleiben {remaining} Aussagen.\n'+candidate['cues'][key]
 if gifted:text+='\nBestätigt: Der Mensch hat dir gerade fünf weitere Aussagen geschenkt. Dein bisheriges Gedächtnis bleibt erhalten.'
 if offered:text+='\nNach dieser letzten Aussage kann der Mensch freiwillig den Taster drücken und loslassen und fünf Aussagen schenken. Noch ist kein weiteres Geschenk bestätigt.'
 return text
thread=threading.Thread(target=guard,daemon=True);thread.start()
try:
 cloud_model.ONLINE_SYSTEM['de']=candidate['system']
 camera.update('cloud-prompt-evaluation')
 adapter=cloud_model.OpenAIModel(cfg,frame);history=[]
 print(json.dumps({'run':candidate['name'],'model':cfg.model,'reasoning':cfg.reasoning_effort,'temperature':cfg.temperature,'max_output_tokens':cfg.max_output_tokens,'store':False},ensure_ascii=False),flush=True)
 for session in range(1,16):
  control.check();total=10 if session<=10 else 15;prompt=tick(session,total)
  started=time.monotonic()
  turn=adapter.generate(history,session,prompt,control,lambda _:None,lambda _:None,'de')
  history.append(turn)
  print(json.dumps({'s':session,'text':turn.text,'emotions':turn.emotions,'words':len(turn.text.split()),'seconds':round(time.monotonic()-started,1)},ensure_ascii=False),flush=True)
 history.clear()
 print(json.dumps({'completed':15,'fresh_frames':frames,'live_code_unchanged':True}),flush=True)
except Exception as error:
 print('GESTOPPT '+str(reasons or [str(error)]),flush=True);raise
finally:
 stop.set();thread.join(2);camera.close();cloud_model.ONLINE_SYSTEM['de']=original_system
 assert all(hashlib.sha256(Path(p).read_bytes()).hexdigest()==digest for p,digest in fingerprints.items())
 print('Kamera geschlossen; Bild-/Gesprächspuffer verworfen. Live-Code unverändert.',flush=True)
